import numpy as np
import re
import json

# Parse numbers from raw input
tokens = [t for t in re.split(r'[\s,;]+', raw_input.strip()) if t]
data = np.array([float(x) for x in tokens])

mean_val = float(np.mean(data))
std_val = float(np.std(data))
var_val = float(np.var(data))
min_val = float(np.min(data))
max_val = float(np.max(data))

print(f"> NumPy array initialized with {len(data)} items.")
print(f"> Mean (np.mean)               : {mean_val:.6f}")
print(f"> Standard Deviation (np.std) : {std_val:.6f}")
print(f"> Variance (np.var)           : {var_val:.6f}")
print(f"> Range                       : [{min_val:.4f}, {max_val:.4f}]")

results = {
    "count": int(len(data)),
    "mean": mean_val,
    "std": std_val,
    "var": var_val,
    "min": min_val,
    "max": max_val
}

json.dumps(results)
